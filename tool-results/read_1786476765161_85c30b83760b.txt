    60→    60→/* ============================================================
    61→    61→   Image Lightbox with Zoom + Pan + Navigation
    62→    62→   ============================================================ */
    63→    63→
    64→    64→function ImageLightbox({ images, startIndex, onClose }: {
    65→    65→  images: { src: string; label: string }[];
    66→    66→  startIndex: number;
    67→    67→  onClose: () => void;
    68→    68→}) {
    69→    69→  const [idx, setIdx] = useState(startIndex);
    70→    70→  const [scale, setScale] = useState(1);
    71→    71→  const [pos, setPos] = useState({ x: 0, y: 0 });
    72→    72→  const [dragging, setDragging] = useState(false);
    73→    73→  const [dragStart, setDragStart] = useState({ x: 0, y: 0 });
    74→    74→  const containerRef = useRef<HTMLDivElement>(null);
    75→    75→
    76→    76→  const [rotation, setRotation] = useState(0);
    77→    77→  const go = (i: number) => { setIdx(i); setScale(1); setPos({ x: 0, y: 0 }); setRotation(0); };
    78→    78→  const prev = () => go(idx > 0 ? idx - 1 : images.length - 1);
    79→    79→  const next = () => go(idx < images.length - 1 ? idx + 1 : 0);
    80→    80→  const rotateLeft = () => setRotation((r) => r - 90);
    81→    81→  const rotateRight = () => setRotation((r) => r + 90);
    82→    82→
    83→    83→  const handleWheel = (e: React.WheelEvent) => {
    84→    84→    e.preventDefault();
    85→    85→    setScale((s) => Math.min(Math.max(s + (e.deltaY > 0 ? -0.15 : 0.15), 0.3), 5));
    86→    86→  };
    87→    87→
    88→    88→  const handleMouseDown = (e: React.MouseEvent) => {
    89→    89→    if (scale <= 1) return;
    90→    90→    setDragging(true);
    91→    91→    setDragStart({ x: e.clientX - pos.x, y: e.clientY - pos.y });
    92→    92→  };
    93→    93→  const handleMouseMove = (e: React.MouseEvent) => {
    94→    94→    if (!dragging) return;
    95→    95→    setPos({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y });
    96→    96→  };
    97→    97→  const handleMouseUp = () => setDragging(false);
    98→    98→
    99→    99→  // Touch support
   100→   100→  const lastTouch = useRef({ x: 0, y: 0, dist: 0 });
   101→   101→  const handleTouchStart = (e: React.TouchEvent) => {
   102→   102→    if (e.touches.length === 2) {
   103→   103→      const dx = e.touches[0].clientX - e.touches[1].clientX;
   104→   104→      const dy = e.touches[0].clientY - e.touches[1].clientY;
   105→   105→      lastTouch.current.dist = Math.sqrt(dx * dx + dy * dy);
   106→   106→    } else if (e.touches.length === 1 && scale > 1) {
   107→   107→      setDragging(true);
   108→   108→      setDragStart({ x: e.touches[0].clientX - pos.x, y: e.touches[0].clientY - pos.y });
   109→   109→    }
   110→   110→  };
   111→   111→  const handleTouchMove = (e: React.TouchEvent) => {
   112→   112→    if (e.touches.length === 2) {
   113→   113→      const dx = e.touches[0].clientX - e.touches[1].clientX;
   114→   114→      const dy = e.touches[0].clientY - e.touches[1].clientY;
   115→   115→      const dist = Math.sqrt(dx * dx + dy * dy);
   116→   116→      const delta = (dist - lastTouch.current.dist) * 0.01;
   117→   117→      setScale((s) => Math.min(Math.max(s + delta, 0.3), 5));
   118→   118→      lastTouch.current.dist = dist;
   119→   119→    } else if (dragging && e.touches.length === 1) {
   120→   120→      setPos({ x: e.touches[0].clientX - dragStart.x, y: e.touches[0].clientY - dragStart.y });
   121→   121→    }
   122→   122→  };
   123→   123→  const handleTouchEnd = () => setDragging(false);
   124→   124→
   125→   125→  useEffect(() => {
   126→   126→    const handleKey = (e: KeyboardEvent) => {
   127→   127→      if (e.key === 'Escape') onClose();
   128→   128→      if (e.key === 'ArrowLeft') prev();
   129→   129→      if (e.key === 'ArrowRight') next();
   130→   130→      if (e.key === '+' || e.key === '=') setScale((s) => Math.min(s + 0.2, 5));
   131→   131→      if (e.key === '-') setScale((s) => Math.max(s - 0.2, 0.3));
   132→   132→      if (e.key === '0') { setScale(1); setPos({ x: 0, y: 0 }); setRotation(0); }
   133→   133→      if (e.key === 'r' || e.key === 'R') setRotation((r) => r + 90);
   134→   134→      if (e.key === 'e' || e.key === 'E') setRotation((r) => r - 90);
   135→   135→    };
   136→   136→    window.addEventListener('keydown', handleKey);
   137→   137→    return () => window.removeEventListener('keydown', handleKey);
   138→   138→  });
   139→   139→
   140→   140→  if (images.length === 0) return null;
   141→   141→
   142→   142→  return (
   143→   143→    <div className="fixed inset-0 z-[9999] bg-black/95 flex flex-col" ref={containerRef}>
   144→   144→      {/* Top bar */}
   145→   145→      <div className="flex items-center justify-between px-4 py-3 shrink-0">
   146→   146→        <div className="text-white text-sm">
   147→   147→          <span className="text-gray-400">{idx + 1} / {images.length}</span>
   148→   148→          <span className="ml-3">{images[idx].label}</span>
   149→   149→        </div>
   150→   150→        <div className="flex items-center gap-2">
   151→   151→          <button onClick={() => { setScale(1); setPos({ x: 0, y: 0 }); setRotation(0); }} className="text-gray-400 hover:text-white text-xs border border-gray-600 px-2 py-1 rounded">Reset</button>
   152→   152→          <button onClick={rotateLeft} className="text-gray-400 hover:text-white text-sm border border-gray-600 px-2 py-1 rounded" title="Girar izquierda (E)">&#8634;</button>
   153→   153→          <button onClick={rotateRight} className="text-gray-400 hover:text-white text-sm border border-gray-600 px-2 py-1 rounded" title="Girar derecha (R)">&#8635;</button>
   154→   154→          <button onClick={() => setScale((s) => Math.min(s + 0.3, 5))} className="text-gray-400 hover:text-white text-lg w-8 h-8 border border-gray-600 rounded" title="Zoom +">+</button>
   155→   155→          <button onClick={() => setScale((s) => Math.max(s - 0.3, 0.3))} className="text-gray-400 hover:text-white text-lg w-8 h-8 border border-gray-600 rounded" title="Zoom -">-</button>
   156→   156→          <button onClick={onClose} className="text-gray-400 hover:text-white text-2xl ml-2">&times;</button>
   157→   157→        </div>
   158→   158→      </div>
   159→   159→
   160→   160→      {/* Main image area */}
   161→   161→      <div
   162→   162→        className="flex-1 flex items-center justify-center overflow-hidden relative cursor-grab active:cursor-grabbing"
   163→   163→        onWheel={handleWheel}
   164→   164→        onMouseDown={handleMouseDown}
   165→   165→        onMouseMove={handleMouseMove}
   166→   166→        onMouseUp={handleMouseUp}
   167→   167→        onMouseLeave={handleMouseUp}
   168→   168→        onTouchStart={handleTouchStart}
   169→   169→        onTouchMove={handleTouchMove}
   170→   170→        onTouchEnd={handleTouchEnd}
   171→   171→        onClick={(e) => {
   172→   172→          // If single click and not dragging, close
   173→   173→          if (!dragging && scale <= 1) onClose();
   174→   174→        }}
   175→   175→      >
   176→   176→        {/* eslint-disable-next-line @next/next/no-img-element */}
   177→   177→        <img
   178→   178→          src={images[idx].src}
   179→   179→          alt={images[idx].label}
   180→   180→          draggable={false}
   181→   181→          className="select-none"
   182→   182→          style={{
   183→   183→            transform: `translate(${pos.x}px, ${pos.y}px) scale(${scale}) rotate(${rotation}deg)`,
   184→   184→            transition: dragging ? 'none' : 'transform 0.15s ease-out',
   185→   185→            maxHeight: '100%',
   186→   186→            maxWidth: '100%',
   187→   187→            objectFit: 'contain',
   188→   188→          }}
   189→   189→        />
   190→   190→      </div>
   191→   191→
   192→   192→      {/* Navigation arrows */}
   193→   193→      {images.length > 1 && (
   194→   194→        <>
   195→   195→          <button onClick={(e) => { e.stopPropagation(); prev(); }} className="absolute left-3 top-1/2 -translate-y-1/2 text-white/70 hover:text-white text-4xl bg-black/30 hover:bg-black/50 w-12 h-12 rounded-full flex items-center justify-center transition-colors">&lsaquo;</button>
   196→   196→          <button onClick={(e) => { e.stopPropagation(); next(); }} className="absolute right-3 top-1/2 -translate-y-1/2 text-white/70 hover:text-white text-4xl bg-black/30 hover:bg-black/50 w-12 h-12 rounded-full flex items-center justify-center transition-colors">&rsaquo;</button>
   197→   197→        </>
   198→   198→      )}
   199→   199→
   200→   200→      {/* Thumbnail strip */}
   201→   201→      {images.length > 1 && (
   202→   202→        <div className="flex gap-2 p-3 overflow-x-auto shrink-0 bg-black/50 justify-center">
   203→   203→          {images.map((img, i) => (
   204→   204→            <button
   205→   205→              key={i}
   206→   206→              onClick={(e) => { e.stopPropagation(); go(i); }}
   207→   207→              className={`shrink-0 w-16 h-16 rounded border-2 overflow-hidden ${i === idx ? 'border-blue-400 ring-2 ring-blue-400/50' : 'border-gray-600 opacity-60 hover:opacity-100'}`}
   208→   208→            >
   209→   209→              {/* eslint-disable-next-line @next/next/no-img-element */}
   210→   210→              <img src={img.src} alt={img.label} className="w-full h-full object-cover" />
   211→   211→            </button>
   212→   212→          ))}
   213→   213→        </div>
   214→   214→      )}
   215→   215→
   216→   216→      {/* Instructions */}
   217→   217→      <div className="text-center text-gray-500 text-[10px] pb-2 shrink-0">
   218→   218→        Rueda mouse / pellizco para zoom &middot; Arrastrar para mover &middot; Flechas para navegar &middot; R / E para girar &middot; Esc para cerrar
   219→   219→      </div>
   220→   220→    </div>
   221→   221→  );
   222→   222→}

   227→   227→
   228→   228→function ThumbGallery({ docs, onOpen }: { docs: { label: string; url?: string }[]; onOpen: (i: number) => void }) {
   229→   229→  const validDocs = docs.filter((d) => d.url && d.url.length > 5);
   230→   230→  if (validDocs.length === 0) return <span className="text-gray-300 text-xs">Sin docs</span>;
   231→   231→  return (
   232→   232→    <div className="flex gap-1 flex-wrap">
   233→   233→      {validDocs.map((d, i) => {
   234→   234→        // Find the real index in the full docs array
   235→   235→        const realIdx = docs.findIndex((dd) => dd.label === d.label);
   236→   236→        return (
   237→   237→          <button
   238→   238→            key={d.label}
   239→   239→            onClick={(e) => { e.stopPropagation(); onOpen(realIdx); }}
   240→   240→            className="relative group w-8 h-8 rounded border border-gray-200 overflow-hidden bg-gray-50 hover:border-blue-400 hover:ring-2 hover:ring-blue-200 transition-all shrink-0"
   241→   241→            title={d.label}
   242→   242→          >
   243→   243→            {/* eslint-disable-next-line @next/next/no-img-element */}
   244→   244→            <img src={d.url} alt={d.label} className="w-full h-full object-cover" />
   245→   245→          </button>
   246→   246→        );
   247→   247→      })}
   248→   248→    </div>
   249→   249→  );
   250→   250→}

   355→   355→function SocioDetail({ socio, onClose }: { socio: SocioRecord; onClose: () => void }) {
